# function definitions

function dirf() {
    argcount=$#

    declare -A dcmds
    dcmds[0]="list entries (cmdnum)"
    dcmds[3]="choose dir (cmdnum)" 
    dcmds[5]="add curr dir (cmdnum)"
    dcmds[7]="add named dir (cmdnum,dir)"
    dcmds[9]="switch to dir num (cmdnum,num)"

    argc=$argcount
    cmdval=$1
    if [ $argcount -eq 0 ]; then
      cmdval=-1
    elif [[ $cmdval = "h" || $cmdval =~ ^-h ]]; then 
        skeys=$(printf "%s\n" "${!dcmds[@]}" | sort -n)    
        for key in $skeys; do
            val="${dcmds[$key]}"
            echo "$key => $val"
        done
        return
    fi
    ### main functional handlers below
    if [ $cmdval -eq 0 ]; then
      echo "${dcmds[$cmdval]}"
      dirs -v
    elif [ $cmdval -eq 3 ]; then
      echo "${dcmds[$cmdval]}"
        dirs -v
        echo "Choose an entry:"
        read unum
        bIsNum=$(echo $unum | awk 'function isnum(x){return(x==x+0)} { if(isnum($0)){print 1}else {print 0} }' )
        if [ $bIsNum -eq 1 ]; then
            pushd +$unum
        else 
            echo "error, need a number, received $unum"
        fi  
    elif [ $cmdval -eq 5  ]; then
      echo "${dcmds[$cmdval]}"
      pushd -n "$(pwd)"
      dirs -v
    elif [ $cmdval -eq 7 ]; then
      echo "${dcmds[$cmdval]}"
      if [ $argc -eq 2 ]; then
        pushd -n "$2"
      else
        echo "need two parameters"
     fi
    elif [ $cmdval -eq 9 ]; then
        echo "${dcmds[$cmdval]}"
        if [ $argc -eq 2 ]; then
            bIsNum=$(echo $2 | awk 'function isnum(x){return(x==x+0)} { if(isnum($0)){print 1}else {print 0} }' )
            if [ $bIsNum -eq 1 ]; then
                pushd +$2
            else 
                echo "error, need a number, received \"$2\""
            fi          
        else
            echo "need two parameters"
        fi
        
    else
      dirs -v
    fi
   
}

export -f dirf
